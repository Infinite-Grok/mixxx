cmake_minimum_required(VERSION 3.16)
project(usbdirect LANGUAGES CXX)

# Standalone transport library (no Mixxx dependency). libusb 1.0.29 headers/archive come from
# LIBUSB_PREFIX (the same prefix tools/android-usb-probe/build.sh uses).
set(LIBUSB_PREFIX "" CACHE PATH "libusb install prefix (include/libusb-1.0, lib/libusb-1.0.a)")
option(USBDIRECT_BUILD_TESTS "Build host tests (use test.sh for the sanitizer run)" OFF)

add_library(usbdirect STATIC src/usb_audio_out.cpp src/device_profile.cpp src/engine_pump.cpp)
target_compile_features(usbdirect PUBLIC cxx_std_17)
target_include_directories(usbdirect PUBLIC include)
target_include_directories(usbdirect PRIVATE ${LIBUSB_PREFIX}/include/libusb-1.0)
target_compile_options(usbdirect PRIVATE -Wall -Wextra -Werror)
if(CMAKE_CROSSCOMPILING AND EXISTS "${LIBUSB_PREFIX}/lib/libusb-1.0.a")
  target_link_libraries(usbdirect PUBLIC "${LIBUSB_PREFIX}/lib/libusb-1.0.a")
endif()
find_package(Threads)
if(Threads_FOUND)
  target_link_libraries(usbdirect PUBLIC Threads::Threads)
endif()

if(USBDIRECT_BUILD_TESTS)
  # Tests link the fake libusb instead of the real archive.
  foreach(t transport_core_test depth_test fifo_test)
    add_executable(${t} tests/${t}.cpp)
    target_include_directories(${t} PRIVATE include)
    target_link_libraries(${t} PRIVATE Threads::Threads)
  endforeach()
  add_executable(device_profile_test tests/device_profile_test.cpp src/device_profile.cpp)
  target_include_directories(device_profile_test PRIVATE include)
  add_executable(usb_audio_out_test tests/usb_audio_out_test.cpp tests/fake_libusb.cpp src/usb_audio_out.cpp src/device_profile.cpp)
  add_executable(engine_pump_test tests/engine_pump_test.cpp tests/fake_libusb.cpp src/usb_audio_out.cpp src/device_profile.cpp src/engine_pump.cpp)
  target_include_directories(engine_pump_test PRIVATE include ${LIBUSB_PREFIX}/include/libusb-1.0)
  target_link_libraries(engine_pump_test PRIVATE Threads::Threads)
  target_include_directories(usb_audio_out_test PRIVATE include ${LIBUSB_PREFIX}/include/libusb-1.0)
  target_link_libraries(usb_audio_out_test PRIVATE Threads::Threads)
  enable_testing()
  foreach(t transport_core_test depth_test fifo_test device_profile_test usb_audio_out_test engine_pump_test)
    add_test(NAME ${t} COMMAND ${t} WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
  endforeach()
endif()
